% Standalone preprint presentation.
% The profile uses a quiet journal vocabulary rather than a publisher or
% institutional imitation: serif manuscript typography, fine rules, and one
% restrained accent for navigation and metadata.

  \thispagestyle{preprintfirst}%
  \begingroup
  \noindent

  {\sffamily\fontsize{8.7}{10.5}\selectfont\color{papermuted}%
    \ifdefempty{\paperbrand}{}{\paperbrand\preprintseparator}%
    \ifdefempty{\paperversion}{}{\paperversion}%
    \ifdefempty{\paperdate}{}{\ifdefempty{\paperversion}{}{\preprintseparator}\paperdate}%
  }%
  \par\vspace{0.48em}
  \preprinttoprule
  \par\vspace{1.38em}
  {\raggedright\bfseries\fontsize{25}{28.5}\selectfont\color{paperink}\papertitle\par}
  \ifdefempty{\papersubtitle}{}{%
    \vspace{0.42em}
    {\raggedright\fontsize{13}{16}\selectfont\color{papermuted}\papersubtitle\par}%
  }
  \vspace{0.95em}
  {\raggedright\bfseries\fontsize{11.1}{13.6}\selectfont\color{paperink}\paperauthors\par}
  \ifdefempty{\paperaffiliations}{}{%
    \vspace{0.38em}
    {\raggedright\fontsize{9.2}{11.6}\selectfont\color{papermuted}\paperaffiliations\par}%
  }
  \preprintmetadatafalse
  \vspace{0.64em}
  {\raggedright\sffamily\fontsize{8.45}{10.6}\selectfont\color{papermuted}%
    \ifdefempty{\correspondingemail}{}{%
      \textbf{Correspondence}\enspace\href{mailto:\correspondingemail}{\nolinkurl{\correspondingemail}}%
      \preprintmetadatatrue
    }%
    \ifdefempty{\paperrepositoryurl}{}{%
      \ifpreprintmetadata\preprintseparator\fi
      \textbf{Code}\enspace\href{\paperrepositoryurl}{\nolinkurl{\paperrepositorylabel}}%

    \ifdefempty{\papercodelicense}{}{%
      \ifpreprintmetadata\preprintseparator\fi

      \preprintmetadatatrue

    \par
  }
  \endgroup
